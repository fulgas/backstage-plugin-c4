> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Mermaid flowchart renderer with a proper C4 React renderer (React Flow + dagre), wire drill-down navigation (level tabs + click-to-navigate) into `DiagramView`, and improve the `C4Page` table with a level badge column, a "Part of" column, and checkbox sidebar filters.

**Architecture:** New `@fulgas/plugin-c4-renderer-react` package provides `ReactC4Renderer` with custom C4-shaped nodes. `c4-common` gains `C4RenderOptions` (optional `onNodeClick` callback) and `parentName` on `C4View`. `DiagramView` wires level tabs + navigation; `C4Page` gets richer columns and sidebar filters.

**Tech Stack:** React Flow (`reactflow`), dagre (`@dagrejs/dagre`), Backstage core-components, Material UI v4, TypeScript, Jest + Testing Library.

- [ ] **Step 4: Add `attachParentNames` and update `getViews` / `getEntityViews` in `ModelStore.ts`**

Add this private method to `ModelStore` (place it before `rowToView`):

```ts
private async attachParentNames(views: C4View[]): Promise<C4View[]> {
  const parentRefs = [
    ...new Set(views.map(v => v.parentRef).filter((r): r is string => !!r)),
  ];
  if (parentRefs.length === 0) return views;
  const elements = await this.db('c4_elements')
    .whereIn('id', parentRefs)
    .select('id', 'name');
  const nameMap = new Map<string, string>(elements.map((e: any) => [e.id as string, e.name as string]));
  return views.map(v =>
    v.parentRef && nameMap.has(v.parentRef)
      ? { ...v, parentName: nameMap.get(v.parentRef) }
      : v,
  );
}
```

Update `getViews` to call it (replace the existing method):

```ts
async getViews(opts?: { type?: C4ViewType; entityRef?: string }): Promise<C4View[]> {
  let query = this.db('c4_views');
  if (opts?.type) query = query.where({ type: opts.type });
  if (opts?.entityRef) query = query.where({ entity_ref: opts.entityRef });
  const rows = await query;
  const views = rows.map(this.rowToView);
  const withNames = await this.attachParentNames(views);
  if (opts?.entityRef) return withNames;
  return dedupeViews(withNames);
}
```

Update `getEntityViews` (replace the existing method):

```ts
async getEntityViews(entityRef: string): Promise<C4View[]> {
  const rows = await this.db('c4_views').where({ entity_ref: entityRef });
  const views = rows.map(this.rowToView);
  return this.attachParentNames(views);
}
```

```tsx
import React from 'react';
import {
  BaseEdge,
  EdgeLabelRenderer,
  EdgeProps,
  getBezierPath,
} from 'reactflow';

export type C4EdgeData = { description?: string; technology?: string };

export function C4Edge({
  id,
  sourceX,
  sourceY,
  targetX,
  targetY,
  sourcePosition,
  targetPosition,
  data,
}: EdgeProps<C4EdgeData>) {
  const [edgePath, labelX, labelY] = getBezierPath({
    sourceX,
    sourceY,
    sourcePosition,
    targetX,
    targetY,
    targetPosition,
  });
  const hasLabel = data?.description || data?.technology;

  return (
    <>
      <BaseEdge
        id={id}
        path={edgePath}
        style={{ stroke: '#888', strokeWidth: 1.5 }}
      />
      {hasLabel && (
        <EdgeLabelRenderer>
          <div
            className="nodrag nopan"
            style={{
              position: 'absolute',
              transform: `translate(-50%, -50%) translate(${labelX}px,${labelY}px)`,
              background: 'rgba(255,255,255,0.9)',
              border: '1px solid #ddd',
              borderRadius: 3,
              padding: '2px 6px',
              fontSize: 10,
              color: '#444',
              pointerEvents: 'none',
              backdropFilter: 'blur(2px)',
            }}
          >
            {data?.description && <div>{data.description}</div>}
            {data?.technology && (
              <div style={{ opacity: 0.7, fontSize: 9 }}>{data.technology}</div>
            )}
          </div>
        </EdgeLabelRenderer>
      )}
    </>
  );
}
```

```ts
import dagre from '@dagrejs/dagre';
import type { Edge, Node } from 'reactflow';

export const NODE_WIDTH = 160;
export const NODE_HEIGHT = 80;

export function applyDagreLayout(
  nodes: Node[],
  edges: Edge[],
  direction: 'TB' | 'LR' = 'TB',
): Node[] {
  const g = new dagre.graphlib.Graph();
  g.setDefaultEdgeLabel(() => ({}));
  g.setGraph({ rankdir: direction, nodesep: 60, ranksep: 100 });

  for (const node of nodes) {
    g.setNode(node.id, { width: NODE_WIDTH, height: NODE_HEIGHT });
  }
  for (const edge of edges) {
    if (g.hasNode(edge.source) && g.hasNode(edge.target)) {
      g.setEdge(edge.source, edge.target);
    }
  }

  dagre.layout(g);

  return nodes.map(node => {
    const { x, y } = g.node(node.id);
    return {
      ...node,
      position: { x: x - NODE_WIDTH / 2, y: y - NODE_HEIGHT / 2 },
    };
  });
}
```

```ts
import type { C4Container, C4ViewModel } from '@fulgas/plugin-c4-common';
import type { Edge, Node } from 'reactflow';
import type { C4EdgeData } from './edges/C4Edge';
import { applyDagreLayout, NODE_HEIGHT, NODE_WIDTH } from './layout';
import type { C4NodeData } from './types';

const BOUNDARY_PADDING = 30;
const BOUNDARY_PADDING_TOP = 40;

export function viewModelToFlow(
  vm: C4ViewModel,
  onNodeClick?: (entityRef: string) => void,
): { nodes: Node<C4NodeData>[]; edges: Edge<C4EdgeData>[] } {
  const { view, model } = vm;
  const included = new Set(view.entityRefs);
  const internal = new Set(view.internalRefs ?? []);
  const includedRels = new Set(view.relationshipIds);

  const rawNodes: Node<C4NodeData>[] = [];

  for (const p of model.persons.filter(x => included.has(x.id))) {
    rawNodes.push({
      id: p.id,
      type: 'personNode',
      position: { x: 0, y: 0 },
      data: { label: p.name, stereotype: '[Person]', onNodeClick },
    });
  }

  for (const s of model.systems.filter(x => included.has(x.id))) {
    const isExt = !internal.has(s.id) && internal.size > 0;
    rawNodes.push({
      id: s.id,
      type: isExt ? 'externalNode' : 'systemNode',
      position: { x: 0, y: 0 },
      data: {
        label: s.name,
        stereotype: isExt ? '[External System]' : '[Software System]',
        catalogEntityRef: s.catalogEntityRef,
        onNodeClick,
      },
    });
  }

  for (const c of model.containers.filter(x => included.has(x.id))) {
    const isExt = !internal.has(c.id) && internal.size > 0;
    rawNodes.push({
      id: c.id,
      type: isExt ? 'externalNode' : 'containerNode',
      position: { x: 0, y: 0 },
      data: {
        label: c.name,
        stereotype: containerStereotype(c, isExt),
        catalogEntityRef: c.catalogEntityRef,
        onNodeClick,
      },
    });
  }

  const rels =
    includedRels.size > 0
      ? model.relationships.filter(r => includedRels.has(r.id))
      : model.relationships.filter(
          r => included.has(r.sourceId) && included.has(r.targetId),
        );

  const edges: Edge<C4EdgeData>[] = rels
    .filter(r => included.has(r.sourceId) && included.has(r.targetId))
    .map(r => ({
      id: r.id,
      source: r.sourceId,
      target: r.targetId,
      type: 'c4Edge',
      data: { description: r.description, technology: r.technology },
    }));

  const direction = view.type === 'landscape' ? 'LR' : 'TB';
  const laidOut = applyDagreLayout(rawNodes, edges, direction);

  const nodes: Node[] = [...laidOut];

  if (internal.size > 0) {
    const internalNodes = laidOut.filter(n => internal.has(n.id));
    if (internalNodes.length > 0) {
      const minX =
        Math.min(...internalNodes.map(n => n.position.x)) - BOUNDARY_PADDING;
      const minY =
        Math.min(...internalNodes.map(n => n.position.y)) -
        BOUNDARY_PADDING_TOP;
      const maxX =
        Math.max(...internalNodes.map(n => n.position.x + NODE_WIDTH)) +
        BOUNDARY_PADDING;
      const maxY =
        Math.max(...internalNodes.map(n => n.position.y + NODE_HEIGHT)) +
        BOUNDARY_PADDING;
      nodes.unshift({
        id: '__boundary__',
        type: 'boundaryGroup',
        position: { x: minX, y: minY },
        style: { width: maxX - minX, height: maxY - minY, zIndex: -1 },
        data: { label: view.title },
        selectable: false,
        draggable: false,
      });
    }
  }

  return { nodes, edges };
}

function containerStereotype(c: C4Container, isExt: boolean): string {
  if (isExt) return '[External Container]';
  const tech = c.technology ? `: ${c.technology}` : '';
  if (c.subType === 'database') return `[Database${tech}]`;
  if (c.subType === 'queue') return `[Queue${tech}]`;
  return `[Container${tech}]`;
}
```

```tsx
  const { views, loading, building } = useEntityC4Views(kind, namespace, name);
  const [selectedViewId, setSelectedViewId] = useState<string | undefined>(
    () => {
      const params = new URLSearchParams(location.search);
      return params.get('view') ?? undefined;
    },
  );
  const { viewModel, loading: vmLoading, error } = useC4View(selectedViewId);

  useEffect(() => {
    if (views && views.length > 0 && !selectedViewId) {
      setSelectedViewId(bestViewId(kind, views));
    }
  }, [views, kind, selectedViewId]);

  // Keep selectedViewId in sync with the ?view= query param (e.g. direct links)
  useEffect(() => {
    const qview = new URLSearchParams(location.search).get('view');
    if (qview && qview !== selectedViewId) setSelectedViewId(qview);
  }, [location.search]); // eslint-disable-line react-hooks/exhaustive-deps

  if (building) {
    return (
      <EmptyState
        missing="data"
        title="Building diagrams…"
        description="C4 diagrams are being generated. Please refresh when done."
      />
    );
  }
  if (loading) return <Progress />;
  if (!views || views.length === 0) {
    return (
      <EmptyState
        missing="data"
        title="No C4 diagrams"
        description="No C4 diagrams found for this entity."
      />
    );
  }

  const availableTypes = [...new Set(views.map(v => v.type))] as C4ViewType[];
  const selectedView = views.find(v => v.id === selectedViewId);
  const selectedType = selectedView?.type ?? availableTypes[0];

  const handleLevelChange = (level: C4ViewType) => {
    const target = views.find(v => v.type === level);
    if (!target) return;
    setSelectedViewId(target.id);
    navigate(
      { search: `?view=${encodeURIComponent(target.id)}` },
      { replace: true },
    );
  };

  const handleNodeClick = (entityRef: string) => {
    const { kind: k, namespace: ns, name: n } = parseRef(entityRef);
    navigate(`/c4/${ns}/${k}/${n}`);
  };

  return (
    <Box>
      {availableTypes.length > 1 && (
        <C4LevelFilter selected={selectedType} onChange={handleLevelChange} />
      )}
      {selectedType && (
        <Typography
          variant="caption"
          style={{ display: 'block', padding: '4px 0 8px', color: '#888' }}
        >
          Viewing: {LEVEL_LABEL[selectedType] ?? selectedType}
        </Typography>
      )}
      <C4DiagramViewer
        viewModel={viewModel}
        renderer={renderer}
        onNodeClick={handleNodeClick}
        loading={vmLoading}
        error={error}
      />
    </Box>
  );
}

export function DiagramView({
  kind,
  namespace,
  name,
}: {
  kind: string;
  namespace: string;
  name: string;
}) {
  return (
    <SWRConfig value={{ provider: () => new Map() }}>
      <DiagramViewInner
        key={`${kind}/${namespace}/${name}`}
        kind={kind}
        namespace={namespace}
        name={name}
      />
    </SWRConfig>
  );
}
```

```tsx
function parseRef(ref: string): {
  kind: string;
  namespace: string;
  name: string;
} {
  const colonIdx = ref.indexOf(':');
  if (colonIdx < 0) return { kind: 'unknown', namespace: 'default', name: ref };
  const kind = ref.slice(0, colonIdx);
  const rest = ref.slice(colonIdx + 1);
  const slashIdx = rest.indexOf('/');
  if (slashIdx < 0) return { kind, namespace: 'default', name: rest };
  return {
    kind,
    namespace: rest.slice(0, slashIdx),
    name: rest.slice(slashIdx + 1),
  };
}
```
